1. [303. 区域和检索 - 数组不可变](https://leetcode.cn/problems/range-sum-query-immutable/)(*)

   ```c++
   class NumArray {
   public:
       vector<int> prefixSum;
   
       NumArray(vector<int>& nums) {
           int n = nums.size();
           prefixSum.assign(n + 1, 0);
           // 构建前缀和数组
           for (int i = 1; i <= n; i++) {
               prefixSum[i] = prefixSum[i - 1] + nums[i - 1];
           }
       }
       
       int sumRange(int left, int right) {
           return prefixSum[right + 1] - prefixSum[left];
       }
   };
   ```

2. [未排序数组中累加和为给定值的最长子数组长度](https://www.nowcoder.com/practice/36fb0fd3c656480c92b569258a1223d5)(*)

   ```c++
   #include <iostream>
   #include <vector>
   #include <unordered_map>
   
   using namespace std;
   
   enum {
       MAXN = 100001
   };
   
   static vector<int> arr(MAXN);
   static unordered_map<int, int> presum;
   
   int main() {
       int sonsum(vector<int>& arr, unordered_map<int, int>&presum, int n, int k);
       int n, k;
       while (cin >> n >> k) { // n:数组长度; k: 目标和
           vector<int> arr(n);
           for (int i = 0; i < n; i++) {
               cin >> arr[i];
           }
           // 求累加和为k的最长数组长度
           cout << sonsum(arr, presum, n, k) << "\n";
       }
   }
   
   int sonsum(vector<int>& arr, unordered_map<int, int>&presum, int n, int k) {
       presum.clear();
       presum.insert({0, -1}); // 0索引前数组和为0
       int sum = 0, ans = 0;
       for (int i = 0; i < n; i++) {
           sum += arr[i];
           if (presum.count(sum - k)) { // 最早出现满足前缀的下标
               ans = max(ans, i - presum[sum - k]);
           }
           // 最早出现的前缀和，加入map
           if (presum.count(sum) == 0) {
               presum[sum] = i;
           }
       }
       return ans;
   }
   ```

3. [560. 和为 K 的子数组](https://leetcode.cn/problems/subarray-sum-equals-k/)(**)

   ```c++
   class Solution {
   public:
       int subarraySum(vector<int>& nums, int k) {
           unordered_map<int, int> prenums;
           prenums[0] = 1; // 前缀和为0的个数
           int sum = 0, ans = 0;
           for (int i = 0; i < nums.size(); i++) {
               sum += nums[i];
               if (prenums.count(sum - k)) {
                   ans += prenums[sum - k]; // 前缀和为k的数的个数
               }
               if (prenums.count(sum)) {
                   prenums[sum]++;
               } else {
                   prenums[sum] = 1;
               }
           }
           return ans;
       }
   };
   ```

4. [无序数组中最长正负个数相等的子数组](https://www.nowcoder.com/practice/545544c060804eceaed0bb84fcd992fb)(*)

   ```c++
   #include <iostream>
   #include <vector>
   #include <unordered_map>
   
   using namespace std;
   
   enum {
       MAXN = 100001
   };
   
   int main() {
       int subpnequal(vector<int>& arr, unordered_map<int, int>& premap, int n);
       int n;
       static vector<int> arr(MAXN);
       static unordered_map<int, int> premap;
       while (cin >> n) { // 数组长度
           for (int i = 0; i < n; i++) {
               cin >> arr[i];
           }
           cout << subpnequal(arr, premap, n) << "\n";
       }
   }
   
   int subpnequal(vector<int>& arr, unordered_map<int, int>& premap, int n) {
       premap.clear();
       premap.insert({0, -1});
       int ans = 0, sum = 0;
       for (int i = 0; i < n; i++) {
           if (arr[i] > 0) {
               sum++;
           } else if (arr[i] < 0) {
               sum--;
           }
           if (premap.count(sum)) { // 前缀和为sum的索引
               ans = max(ans, i - premap[sum]);
           } else {
               premap.insert({sum, i});
           }
       }
       return ans;
   }
   ```

5. [1124. 表现良好的最长时间段](https://leetcode.cn/problems/longest-well-performing-interval/)(**)

   > 给你一份工作时间表 `hours`，上面记录着某一位员工每天的工作小时数。
   >
   > 我们认为当员工一天中的工作小时数大于 `8` 小时的时候，那么这一天就是「**劳累的一天**」。
   >
   > 所谓「表现良好的时间段」，意味在这段时间内，「劳累的天数」是严格 **大于**「不劳累的天数」。
   >
   > 请你返回「表现良好时间段」的最大长度。

   > 1. 劳累的一天记为1，不劳累记为-1，要求为1的天数严格大于-1的天数的区间;
   > 2. `i` 位置前缀和大于0，则 `0~i` 段是 `表现良好的时间段`，并且是到 `i` 位置的最长满足时间段；
   > 3. `i`  位置前缀和 `prefixsum` 小于等于0，若前面有前缀和为 $prefixsum - 1​$ 的，那从它后一位到 `i` 的和为 1。
   > 4. 并且由于数组只有1和-1，前缀和是连续的，若存在前缀和小于 $prefixsum - 1$的，则之前一定存在前缀和为 $prefixsum - 1$ 的下标。而在哈希表中记录的信息是 `key` 为前缀和，`value` 为最早下标，所以前缀和为$prefixsum - 1$ 的后一位到 `i` 位，就是 `prefixsum` 小于等于0时，`i`位置及以前的最长良好子数组。

   ```c++
   class Solution {
   public:
       int longestWPI(vector<int>& hours) {
           unordered_map<int, int> premap;
           premap.insert({0, -1});
           int sum = 0, ans = 0;
           for (int i = 0; i < hours.size(); i++) {
               // 大于8的视为1，否则为-1，求前缀和
               if (hours[i] > 8) {
                   sum++;
               } else {
                   sum--;
               }
               if (sum > 0) {
                   ans = i + 1;
               } else { // sum<=0
                   if (premap.count(sum - 1)) { // 出现前缀和为sum-1的最早位置
                       ans = max(ans, i - premap[sum - 1]);
                   }
               }
               // 添加前缀和为sum的最早下标
               if (!premap.count(sum)) {
                   premap.insert({sum, i});
               }
           }
           return ans;
       }
   };
   ```

6. [1590. 使数组和能被 P 整除](https://leetcode.cn/problems/make-sum-divisible-by-p/)(**)

   > 给你一个正整数数组 `nums`，请你移除 **最短** 子数组（可以为 **空**），使得剩余元素的 **和** 能被 `p` 整除。 **不允许** 将整个数组都移除。
   >
   > 请你返回你需要移除的最短子数组的长度，如果无法满足题目要求，返回 `-1` 。

   > 相当于找到子数组和模·p的余数和整个数组和模p余数相等的最短子数组

   ```c++
   class Solution {
   public:
       int minSubarray(vector<int>& nums, int p) {
           int mod = 0;
           for (int& x: nums) {
               mod = (mod + x) % p; // nums和模p余数
           }
           if (mod == 0) {
               return 0;
           }
           unordered_map<int, int> modidx; // 记录前缀和模p余数最晚出现位置
           modidx.insert({0, -1});
           int ans = INT_MAX;
           int premod, submod = 0;
           for (int i = 0; i < nums.size(); i++) {
               submod = (submod + nums[i]) % p;
               // 减法同余原理，
               // premod = submod >= mod ? (submod - mod) : (submod + p - mod);
               premod = (submod + p - mod) % p;
               if (modidx.count(premod)) {
                   ans = min(ans, i - modidx[premod]);
               }
               modidx[submod] = i; // 余数的最晚位置
           }
           return ans == nums.size() ? -1 : ans;
       }
   };
   ```

7. [1371. 每个元音包含偶数次的最长子字符串](https://leetcode.cn/problems/find-the-longest-substring-containing-vowels-in-even-counts/)(**)

   > 给你一个字符串 `s` ，请你返回满足以下条件的最长子字符串的长度：每个元音字母，即 'a'，'e'，'i'，'o'，'u' ，在子字符串中都恰好出现了偶数次。

   ```c++
   class Solution {
   public:
       int findTheLongestSubstring(string s) {
           array<int, 32> status; // 5个字母的奇偶状态，5位32个
           status.fill(-2); // 初始化为-2，表示之前没出现过的状态
           status[0] = -1; // 最早全为偶数次下标
           int ans = 0;
           int m, cur = 0;
           for (int i = 0; i < s.size(); i++) {
               m = mapping(s[i]);
               // 当前字母是元音，状态对应位异或1，为元音字母当前奇偶状态
               if (m != -1) {
                   cur ^= 1 << m;
               }
               if (status[cur] != -2) {
                   ans = max(ans, i - status[cur]);
               } else {
                   status[cur] = i; // 最早下标
               }
           }
           return ans;
       }
   
       int mapping(char c) {
           switch (c) {
               case 'a': return 0;
               case 'e': return 1;
               case 'i': return 2;
               case 'o': return 3;
               case 'u': return 4;
               default: return -1;
           }
       }
   };
   ```

1. [304. 二维区域和检索 - 矩阵不可变](https://leetcode.cn/problems/range-sum-query-2d-immutable/)(**)

   ```c++
   class NumMatrix {
   public:
       vector<vector<int>> sumtri;
       NumMatrix(vector<vector<int>>& matrix) {
           int m = matrix.size();
           int n = matrix[0].size();
           sumtri = vector<vector<int>>(m + 1, vector<int>(n + 1, 0));
           for (int i = 1; i <= m; i++) {
               for (int j = 1; j <= n; j++) {
                   sumtri[i][j] = matrix[i - 1][j - 1];
               }
           }
           // 二位前缀和
           for (int i = 1; i <= m; i++) {
               for (int j = 1; j <= n; j++) {
                   sumtri[i][j] += sumtri[i][j - 1] + sumtri[i - 1][j] - sumtri[i - 1][j - 1];
               }
           }
       }
       
       int sumRegion(int a, int b, int c, int d) {
           c++, d++;
           return sumtri[c][d] - sumtri[c][b] - sumtri[a][d] + sumtri[a][b];
       }
   };
   ```

2. [1139. 最大的以 1 为边界的正方形](https://leetcode.cn/problems/largest-1-bordered-square/)(**)

   > 给你一个由若干 `0` 和 `1` 组成的二维网格 `grid`，请你找出边界全部由 `1` 组成的最大 **正方形** 子网格，并返回该子网格中的元素数量。如果不存在，则返回 `0`。

   ```c++
   class Solution {
   public:
       int largest1BorderedSquare(vector<vector<int>>& grid) {
           int m = grid.size();
           int n = grid[0].size();
           // 建立二维前缀树组
           build(grid, m, n);
           if (sum(grid, 0, 0, m - 1, n - 1) == 0) {
               return 0;
           }
           // 找最大合法正方形
           int ans = 1;
           for (int a = 0; a < m; a++) {
               for (int b = 0; b < n; b++) {
                   // 找边长为k的合法正方形
                   for (int c = a + ans, d = b + ans, k = ans + 1; 
                       c < m && d < n; c++, d++, k++) {
                       if (sum(grid, a, b, c, d) - sum(grid, a + 1, b + 1, c - 1, d - 1) 
                       == (k - 1) << 2) {
                           ans = k;
                       }
                   }
               }
           }
           return ans * ans; // 最大合法正方形面积
       }
   
       // 二维前缀和数组
       void build(vector<vector<int>>& grid, int m, int n) {
           for (int i = 0; i < m; i++) {
               for (int j = 0; j < n; j++) {
                   grid[i][j] += get(grid, i - 1, j) + get(grid, i, j - 1) 
                   - get(grid, i - 1, j - 1);
               }
           }
       }
   
       // 防越界
       int get(vector<vector<int>>& grid, int i, int j) {
           if (i < 0 || j < 0) {
               return 0;
           } else {
               return grid[i][j];
           }
       }
   
       int sum(vector<vector<int>>& grid, int a, int b, int c, int d) {
           if (a > c) {
               return 0;
           } else {
               return grid[c][d] - get(grid, c, b - 1) - get(grid, a - 1, d)
               + get(grid, a - 1, b - 1);
           }
           
       }
   };
   ```
