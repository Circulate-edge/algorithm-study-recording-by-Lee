1. [单调栈结构(进阶)](https://www.nowcoder.com/practice/2a2c00e7a88a498693568cef63a4b7bb) (中等)

   > 给定一个可能含有重复值的数组 arr，找到每一个 i 位置左边和右边离 i 位置最近且值比 arr[i] 小的位置。返回所有位置相应的信息。
   >
   > 输入描述：
   >
   > 第一行输入一个数字 n，表示数组 arr 的长度。
   > 以下一行输入 n 个数字，表示数组的值
   >
   > 输出描述：
   >
   > 输出n行，每行两个数字 L 和 R，如果不存在，则值为 -1，下标从 0 开始。

   ```c++
   #include <iostream>
   #include <array>
   
   using namespace std;
   enum {MAXN = 1000001};
   static array<int, MAXN> arr = {};
   static array<int, MAXN> stk = {};
   static array<array<int, 2>, MAXN> ans = {};
   void inst(int n);
   
   int main() {
       int n;
       while (cin >> n) { // arr长度
           for (int i = 0; i < n; i++) {
               cin >> arr[i];
           }
           inst(n); // 单调栈解
           for (int i = 0; i < n; i++) {
               cout << ans[i][0] << ' ' << ans[i][1] << '\n';
           }
       }
   }
   
   void inst(int n) {
       int cur, size = 0;
       // 入栈
       for (int i = 0; i < n; i++) {
           while (size > 0 && arr[i] <= arr[stk[size - 1]]) { // <=都出栈，站内严格单调增加
               // 结算出栈元素
               cur = stk[--size];
               ans[cur][1] = i; // 右边答案,存索引
               ans[cur][0] = size > 0 ? stk[size - 1] : -1; // 左边答案
           }
           stk[size++] = i;
       }
   
       // 决算
       while (size > 0) {
           cur = stk[--size];
           ans[cur][1] = -1; // 清算阶段右边答案不存在
           ans[cur][0] = size > 0 ? stk[size - 1] : -1;
       }
   
       // 右侧可能出现相等，修正右侧答案
       // 从右往左修正，n-1位置的右侧答案一定是-1，不需要修正
       for (int i = n - 2; i >= 0; i--) {
           if (arr[ans[i][1]] == arr[i]) {
               ans[i][1] = ans[ans[i][1]][1]; // 答案更新为右边ans[1]记录的答案
           }
       }
   }
   ```

2. [739. 每日温度](https://leetcode.cn/problems/daily-temperatures/) (medium)

   > 给定一个整数数组 `temperatures` ，表示每天的温度，返回一个数组 `answer` ，其中 `answer[i]` 是指对于第 `i` 天，下一个更高温度出现在几天后。如果气温在这之后都不会升高，请在该位置用 `0` 来代替。

   ```c++
   class Solution {
   public:
       vector<int> dailyTemperatures(vector<int>& temperatures) {
           int n = temperatures.size();
           stack<int> tem;
           vector<int> ans(n, 0);
           for (int i = 0; i < n; i++) {
               int cur;
               // 只用找未来温度更高的日期，严格大于才出栈处理
               while (!tem.empty() && temperatures[i] > temperatures[tem.top()]) {
                   cur = tem.top();
                   tem.pop();
                   ans[cur] = i - cur;
               }
               tem.push(i);
           }
           return ans;
       }
   };
   ```

3. [907. Sum of Subarray Minimums](https://leetcode.com/problems/sum-of-subarray-minimums/) (medium)

   > Given an array of integers arr, find the sum of `min(b)`, where `b` ranges over every (contiguous) subarray of `arr`. Since the answer may be large, return the answer **modulo** `109 + 7`.

   ```c++
   class Solution {
   private:
       static const int MAXN = 30001;
   public:
       inline static vector<int> stk = vector<int>(MAXN, 0);
       int sumSubarrayMins(vector<int>& arr) {
           static int MOD = 1e9 + 7;
           int size = 0;
           long long sum = 0;
           int n = arr.size();
           for (int i = 0; i < n; i++) {
               int cur, l;
               while (size > 0 && arr[i] <= arr[stk[size - 1]]) { // 小于等于都出栈处理
                   cur = stk[--size];
                   l = size > 0 ? stk[size - 1] : -1;
                   // 统计当前元素贡献, 是几个连续子数组的最小和
                   sum = (sum + (long long)arr[cur] * (cur - l) * (i - cur)) % MOD;
               }
               stk[size++] = i;
           }
   
           // 处理站内剩余元素
           while (size > 0) {
               int cur = stk[--size];
               int l = size > 0 ? stk[size - 1] : -1;
               // 统计当前元素贡献, 是几个连续子数组的最小和
               sum = (sum + (long long)arr[cur] * (cur - l) * (n - cur)) % MOD;
           }
           return (int)sum;
       }
   };
   ```

4. [84. 柱状图中最大的矩形](https://leetcode.cn/problems/largest-rectangle-in-histogram/) (hard)

   > 给定 *n* 个非负整数，用来表示柱状图中各个柱子的高度。每个柱子彼此相邻，且宽度为 1 。
   >
   > 求在该柱状图中，能够勾勒出来的矩形的最大面积。
   >
   > **示例 1:**
   >
   > ![img](img/histogram.jpg)
   >
   > **提示：**
   >
   > - `1 <= heights.length <=105`
   > - `0 <= heights[i] <= 104`

   ```c++
   class Solution {
       static const int MAXN = 100001;
       inline static array<int, MAXN> stk = {};
   public:
       int largestRectangleArea(vector<int>& heights) {
           int l, cur, size = 0;
           int n = heights.size();
           int ans = 0;
           for (int i = 0; i < n; i++) {
               // 维护栈严格单调增加
               while (size > 0 && heights[i] <= heights[stk[size - 1]]) { // 和栈顶元素比较
                   cur = stk[--size];
                   l = size > 0 ? stk[size - 1] : -1; // 左侧还有更小元素
                   ans = max(ans, heights[cur] * (i - l - 1)); // 能围成的最大面积
               }
               stk[size++] = i;
           }
           while (size > 0) {
               cur = stk[--size];
               l = size > 0 ? stk[size - 1] : -1; // 左侧还有更小元素
               ans = max(ans, heights[cur] * (n - l - 1)); // 能围成的最大面积
           }
           return ans;
       }
   };
   ```

5. [85. 最大矩形](https://leetcode.cn/problems/maximal-rectangle/) (hard)

   > 给定一个仅包含 `0` 和 `1` 、大小为 `rows x cols` 的二维二进制矩阵，找出只包含 `1` 的最大矩形，并返回其面积。
   >
   > **示例 1：** 
   >
   > ![img](img/1722912576-boIxpm-image.png)
   >
   > **提示：** 
   >
   > - `rows == matrix.length` 
   > - `cols == matrix[0].length` 
   > - ` 1 <= rows, cols <= 200` 
   > - `matrix[i][j]` 为 `'0'` 或 `'1'` 

   > **时间复杂度：** $O(m * n)​$ 
   >
   > * 对 `m` 行压缩，用单调栈求以每行为底的柱状图的最大矩形，过程中遍历n次

   ```c++
   class Solution {
       static const int MAXN = 201;
       inline static array<int, MAXN> stk = {};
       inline static array<int, MAXN> hs = {};
   public:
       int maximalRectangle(vector<vector<char>>& matrix) {
           int m = matrix.size();
           int n = matrix[0].size();
           int ans = 0;
           hs.fill(0);
           for (int c = 0; c < m; c++) { // 以第c行为底压缩数组成直方图
               for (int i = 0; i < n; i++) {
                   hs[i] = matrix[c][i] == '1' ? hs[i] + 1 : 0;
               }
               // 用单调栈
               ans = max(ans, rlarge(n));
           }
           return ans;
       }
   
       int rlarge(int n) {
           int size = 0, cur, l, ans = 0;
           for (int i = 0; i < n; i++) {
               while (size > 0 && hs[i] <= hs[stk[size - 1]]) {
                   cur = stk[--size];
                   l = size > 0 ? stk[size - 1] : -1; // 左侧没有更小的则左侧索引设为-1
                   ans = max(ans, hs[cur] * (i - 1 - l));
               }
               stk[size++] = i; // 索引入栈，对应高度单调增
           }
           // 栈内剩余
           while (size > 0) {
               cur = stk[--size];
               l = size > 0 ? stk[size - 1] : -1;
               ans = max(ans, hs[cur] * (n - 1 - l));
           }
           return ans;
       }
   };
   ```

1. [962. 最大宽度坡](https://leetcode.cn/problems/maximum-width-ramp/) (medium)

   > 给定一个整数数组 `A`，*坡*是元组 `(i, j)`，其中  `i < j` 且 `A[i] <= A[j]`。这样的坡的宽度为 `j - i`。
   >
   > 找出 `A` 中的坡的最大宽度，如果不存在，返回 0 。
   >
   > **提示：**
   >
   > 1. `2 <= A.length <= 50000`
   > 2. `0 <= A[i] <= 50000`

   ```c++
   class Solution {
       static const int MAXN = 50001;
       inline static array<int, MAXN> stk = {};
   public:
       int maxWidthRamp(vector<int>& nums) {
           // 单调栈，0位置已经进栈了，栈内元素严格单调减
           stk.fill(0);
           int n = nums.size();
           int size = 1;
           for (int i = 1; i < n; i++) {
               if (nums[stk[size - 1]] > nums[i]) {
                   stk[size++] = i;
               }   
           }
           // 从后往前遍历
           int ans = 0;
           for (int i = n - 1; i >= 0; i--) {
               while (size > 0 && nums[stk[size - 1]] <= nums[i]) {
                   ans = max(ans, i - stk[--size]);
               }
           }
           return ans;
       }
   };
   ```

2. [316. 去除重复字母](https://leetcode.cn/problems/remove-duplicate-letters/) (medium)

   > 给你一个字符串 `s` ，请你去除字符串中重复的字母，使得每个字母只出现一次。需保证 **返回结果的字典序最小**（要求不能打乱其他字符的相对位置）。
   >
   > **提示：**
   >
   > - `1 <= s.length <= 10^4`
   > - `s` 由小写英文字母组成

   ```c++
   class Solution {
       inline static array<int, 26> cnts = {}; // 词频
       inline static array<bool, 26> ct = {}; // 栈内有没有包含单词
       inline static array<char, 26> stk = {};
   public:
       string removeDuplicateLetters(string s) {
           int size = 0;
           cnts.fill(0);
           ct.fill(false);
           for (char& c: s) { // 统计词频
               cnts[c - 'a']++;
           }
           for (int i = 0; i < s.size(); i++) {
               if (!ct[s[i] - 'a']) { // 栈内没有这个字符
                   // 尽可能维持单调, 除非栈顶的字符后续不会再出现
                   while (size > 0 && stk[size - 1] > s[i] 
                   && cnts[stk[size - 1] - 'a'] > 0) {
                       ct[stk[--size] - 'a'] = false;
                   }
                   stk[size++] = s[i];
                   ct[s[i] - 'a'] = true;
               }
               cnts[s[i] - 'a']--; // 不管栈内有没有这个元素，词频都减少
           }
           string ans = "";
           for (int i = 0; i < size; i++) {
               ans += stk[i];
           }
           return ans;
       }
   };
   ```

3. [大雨吃小鱼_牛客题霸_牛客网](https://www.nowcoder.com/practice/77199defc4b74b24b8ebf6244e1793de) (中等)

   > 现在有N条鱼，每条鱼的体积为Ai，从左到右排成一排。
   >
   > A数组是一个排列。
   >
   > 小明每轮可以执行一次大鱼吃小鱼的操作
   >
   > 一次大鱼吃小鱼的操作：对于每条鱼，它在每一次操作时会吃掉右边比自己小的第一条鱼
   >
   > 值得注意的时，在一次操作中，每条鱼吃比自己小的鱼的时候是同时发生的。
   >
   > 举一个例子，假设现在有三条鱼,体积为分别[5，4，3]，5吃4，4吃3，一次操作后就剩下[5]一条鱼。
   >
   > **输入描述：** 
   >
   > 给定N；
   >
   > 给定A数组
   >
   > １＜＝N＜＝１０＾５
   >
   > １＜＝Ai＜＝Ｎ

   ```c++
   #include <iostream>
   #include <array>
   
   using namespace std;
   
   enum {
       MAXN = 100001
   };
   static array<int, MAXN> arr = {};
   static array<array<int, 2>, MAXN> stk = {};
   int turn(int n);
   
   int main() {
       ios::sync_with_stdio(false);
       cin.tie(nullptr);
   
       int n;
       while (cin >> n) { // 数组大小
           for (int i = 0; i < n; i++) {
               cin >> arr[i]; 
           }
           // 从后遍历，维持单调
           cout << turn(n) << '\n';
       }
   }
   
   int turn(int n) {
       // 从后遍历，维持单调减，也就是鱼不会变
       int size = 0;
       int ans = 0;
       for (int i = n - 1; i >= 0; i--) {
           // 比栈顶大的就决算，大于吃小鱼
           int turns = 0; // 当前轮次
           while (size > 0 && arr[stk[size - 1][0]] < arr[i]) {
               // 轮次计算，因为同一轮是同时进行的,所以只取最大
               turns = max(turns + 1, stk[--size][1]); 
           }
           stk[size][0] = i;
           stk[size++][1] = turns;
           ans = max(ans, turns);
       }
       return ans;
   }
   ```

4. [2289. 使数组按非递减顺序排列](https://leetcode.cn/problems/steps-to-make-array-non-decreasing/) (medium)

   > 给你一个下标从 **0** 开始的整数数组 `nums` 。在一步操作中，移除所有满足 `nums[i - 1] > nums[i]` 的 `nums[i]` ，其中 `0 < i < nums.length` 。
   >
   > 重复执行步骤，直到 `nums` 变为 **非递减** 数组，返回所需执行的操作数。
   >
   > **提示：**
   >
   > - `1 <= nums.length <= 10^5`
   > - `1 <= nums[i] <= 10^9`

   > **思路：** 
   >
   > 1. 从后往前遍历，维持单调栈递减，
   > 2. 栈中相邻的元素必然不是同一轮被消除，所以后面进栈的元素结算前面元素轮次+1；因为能结算当前栈顶元素的一定更大，也能结算栈内后续元素，可以继承后面元素的轮次，并且大鱼吃小鱼可以同时进行，所以记录最大轮次。

   ```c++
   class Solution {
       static const int MAXN = 100001;
       inline static array<array<int, 2>, MAXN> stk;
   public:
       int totalSteps(vector<int>& nums) {
           // 从后往前遍历，维持单调递减，栈中相邻的元素必然不是同一轮被消除
           int size = 0;
           int ans = 0;
           for (int i = nums.size() - 1; i >= 0; i--) {
               int curturns = 0;
               // 比栈顶大的结算栈顶，大鱼吃小鱼
               while (size > 0 && nums[stk[size - 1][0]] < nums[i]) {
                   // 结算时当前轮次+1，但是和后面的其他元素结算同时
                   // 大的元素能代替后面的元素结算，故轮次取最大值
                   curturns = max(curturns + 1, stk[--size][1]);
               }
               stk[size][0] = i;
               stk[size++][1] = curturns; // 记录当前元素结算后面的花了几轮
               ans = max(ans, curturns);
           }
           return ans;
       }
   };
   ```

5. [1504. 统计全 1 子矩形](https://leetcode.cn/problems/count-submatrices-with-all-ones/) (medium)

   > 给你一个 `m x n` 的二进制矩阵 `mat` ，请你返回有多少个 **子矩形** 的元素全部都是 1 。

   ```c++
   class Solution {
       static const int MAXN = 151; // 列数
       inline static array<int, MAXN> height = {}; // 高度直方图
       inline static array<int, MAXN> stk = {}; 
   public:
       int numSubmat(vector<vector<int>>& mat) {
           int m = mat.size(); // 行数
           int n = mat[0].size(); // 列数
           int ans = 0;
           height.fill(0); // 初始化
           for (int i = 0; i < m; i++) {
               // 以i行为底生成为直方图, 统计子矩形
               for (int j = 0; j < n; j++) {
                   height[j] = mat[i][j] == 0 ? 0 : height[j] + 1;
               }
               ans += c(n);
           }
           return ans;
       }
   
       // 统计子矩形
       int c(int n) {
           int size = 0, res = 0, cur;
           for (int i = 0; i < n; i++) {
               // 比栈顶小的会使栈顶元素结算, 相等的可以压栈, 不影响
               while (size > 0 && height[stk[size - 1]] > height[i]) {
                   // 出栈结算
                   cur = stk[--size];
                   int l = size > 0 ? stk[size - 1] : -1;
                   // 统计高度为max(height[l], height[r])~height[cur]的子矩形个数
                   int bottom = max(l == -1 ? 0 : height[l], height[i]);
                   int len = (i - l - 1);
                   res += (height[cur] - bottom) * len * (len + 1) / 2;
               }
               stk[size++] = i;
           }
   
           // 计算栈内剩余元素
           while (size > 0) {
               cur = stk[--size];
               int l = size > 0 ? stk[size - 1] : -1;
               int bottom = l == -1 ? 0 : height[l]; // 右边不存在更低元素, 视为0
               int len = n - l - 1;
               res += (height[cur] - bottom) * len * (len + 1) / 2;
           }
           return res;
       }
   };
   ```
