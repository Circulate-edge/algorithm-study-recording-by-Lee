1. [9. 回文数](https://leetcode.cn/problems/palindrome-number/)(*)

   ```c++
   class Solution {
   public:
       bool isPalindrome(int x) {
           if (x < 0) return false;
           int offset = 1; // 最高位
           while (x / offset >= 10) {
               offset *= 10;
           }
           // 首尾相等
           while (x > 0) {
               if (x / offset != x % 10) return false;
               x = (x % offset) / 10; // 掐头去尾
               offset /= 100;
           }
           return true;
       }
   };
   ```

   

2. [906. 超级回文数](https://leetcode.cn/problems/super-palindromes/)(***)

   > 如果一个正整数自身是回文数，而且它也是一个回文数的平方，那么我们称这个数为 超级回文数 。
   > 现在，给你两个以字符串形式表示的正整数 left 和 right  ，
   > 统计并返回区间 [left, right] 中的 超级回文数 的数目。
   >
   > - `1 <= left.length, right.length <= 18`
   > - `left` 和 `right` 仅由数字（0 - 9）组成。
   > - `left` 和 `right` 不含前导零。
   > - `left` 和 `right` 表示的整数在区间 `[1, 1018 - 1]` 内。
   > - `left` 小于等于 `right` 。

   ```c++
   class Solution {
   public:
       /**
        * 如果一个正整数自身是回文数，而且它也是一个回文数的平方，那么我们称这个数为 超级回文数 。
        * 现在，给你两个以字符串形式表示的正整数 left 和 right  ，
        * 统计并返回区间 [left, right] 中的 超级回文数 的数目。
        */
       int superpalindromesInRange(string left, string right) {
           long long l = stoll(left);
           long long r = stoll(right);
           long long limit = (long long)sqrt(r); // 种子限制
           long long seed = 1; // 超级回文数种子
           int ans = 0;
           long long odd = 1, even = 1;
           long long pn;
           while (true) {
               odd = genOdd(seed); // 奇数级回文数，范围更小
               even = genEven(seed); // 偶数级回文数
               if (odd > limit && even > limit) break;
   
               if (odd <= limit) {
                   pn = odd * odd;
                   if (check(pn, l, r)) {
                       ans += 1;
                   }
               }
               if (even <= limit) {
                   pn = even * even;
                   // 检查
                   if (check(pn, l, r)) {
                       ans += 1;
                   }
               }
               
               seed++; 
           }
           return ans;
       }
   
       // 生成偶数长度回文数(2n)
       long long genEven(long long seed) {
           long long palin = seed;
           while (seed > 0) {
               palin = palin * 10 + (seed % 10);
               seed /= 10;
           }
           return palin;
       }
   
       // 生成奇数长度回文数(2n - 1)
       long long genOdd(long long seed) {
           long long palin = seed;
           while ((seed /= 10) > 0) {
               palin = palin * 10 + (seed % 10);
           }
           return palin;
       }
   
       // 判断是不是回文数
       bool isPalin(long long num) {
           long long offset = 1; // 数字的最高位
           while ((num / offset) >= 10) {
               offset *= 10;
           }
           while (num > 0) {
               // 首尾比较
               // num:   34543
               // offset:10000
               if ((num / offset) != (num % 10)) {
                   return false;
               }
               num = (num % offset) / 10;
               offset /= 100;
           }
           return true;
       }
   
       bool check(long long num, long long l, long long r) {
           if (num >= l && num <= r) {
               if (isPalin(num)) {
                   return true;
               }
           }
           return false;
       }
   };
   ```
