3. [NOIP 2008 提高组\] 笨小猴   ](https://www.luogu.com.cn/problem/P1125)(*)

   > 假设 `maxn` 是单词中出现次数最多的字母的出现次数，`minn` 是单词中出现次数最少的字母的出现次数，如果` maxn−minn `是一个质数，那么笨小猴就认为这是个 Lucky Word，这样的单词很可能就是正确的答案。

   ```c++
   #include <iostream>
   #include <string>
   #include <unordered_map>
   #include <climits>
   #include <cmath>
   
   using namespace std;
   bool prime_number(int num);
   
   int main() {
       unordered_map<char,int> dc;
       string str;
       cin >> str;
       for (char c: str) {
           if (dc.find(c) == dc.end()) {
               dc[c]=1;
           } else {
               dc[c]++;
           }
       }
       int maxn = INT_MIN;
       int minn = INT_MAX;
       for (const auto& pair: dc) {
           maxn = pair.second > maxn ? pair.second : maxn;
           minn = pair.second < minn ? pair.second : minn;
       }
       int n = !dc.empty() ? maxn-minn : 0;
   
       if (prime_number(n)) {
           cout << "Lucky Word" << '\n' << n;
       } else {
           cout << "No Answer" << '\n' << 0;
       }
   }
   
   bool prime_number(int num) {
       if (num < 2) {
           return false;
       }
       for (int i=2; i <= sqrt(num); i++) {
           if (num % i == 0) {
               return false;
           }
       }
       return true;
   }
   ```

4. [P1957     口算练习题   ](https://www.luogu.com.cn/problem/P1957)(*)

   > **输入格式**
   >
   > 第一行一个整数 *i*。
   >
   > 接着的 *i* 行为需要输入的算式，每行可能有三个数据或两个数据。
   >
   > 若该行为三个数据则第一个数据表示运算类型，a 表示加法运算，b 表示减法运算，c 表示乘法运算，接着的两个数据表示参加运算的运算数。
   >
   > 若该行为两个数据，则表示本题的运算类型与上一题的运算类型相同，而这两个数据为运算数。
   >
   > **输出格式**
   >
   > 输出 2×*i* 行。对于每个输入的算式，输出完整的运算式及结果，第二行输出该运算式的总长度。

   ```c++
   #include <iostream>
   #include <string>
   
   using namespace std;
   
   int main() {
       int i;
       cin >> i;
       string pre;
       for (int a=0; a<i; a++) {
           string f;
           string op;
           int x,y,z;
           cin >> f;
           if (f == "a" || f=="b" || f=="c") {
               // 3参数情况
               pre = f; // renew
               cin >> x >> y;
           } else {
               // 2参数
               x = stoi(f);
               cin >> y;
           }
           if (pre == "a") {
               op = '+';
               z = x + y;
           } else if (pre == "b") {
               op = '-';
               z = x - y;
           } else if (pre == "c") {
               op = '*';
               z = x * y;
           }
           string res = to_string(x)+op+to_string(y)+'='+to_string(z);
           cout << res << '\n';
           cout << res.length() << '\n';
       }
       return 0;
   }
   ```

6. [P5734 【深基6.例6】文字处理软件](https://www.luogu.com.cn/problem/P5734#ide)(*)

   ```c++
   #include <iostream>
   #include <string>
   
   using namespace std;
   
   int main () {
       int q;
       string str;
       cin >> q >> str;
       for (int i=0; i<q; i++) {
           int ty;
           cin >> ty;
           if (ty == 1) {
               // 操作1，尾接
               string ins;
               cin >> ins;
               str += ins;
               cout << str << '\n';
           } else if (ty == 2) {
               // 操作2，从a开始取b个长度
               int a,b;
               cin >> a >> b;
               str = str.substr(a,b);
               cout << str << '\n';
           } else if (ty == 3) {
               // 操作3，a位置前插入
               int a;
               string pc;
               cin >> a >> pc;
               str.insert(a,pc);
               cout << str << '\n';
           } else if (ty == 4) {
               // 操作4，查找
               string sub;
               cin >> sub;
               int pos;
               if (str.find(sub) != string::npos) {
                   pos = (int)str.find(sub);
               } else {
                   pos = -1;
               }
               cout << pos << '\n';
           }
       }
   }
   ```

   

7. [P1308     NOIP 2011 普及组\统计单词数   ](https://www.luogu.com.cn/problem/P1308)(**)

   ```c++
   #include <iostream>
   #include <string>
   
   using namespace std;
   
   int main() {
       string word;
       string file;
       cin >> word;
       // 忽略残留换行符
       cin.ignore();
       int lw = (int)word.size();
       // 转小写
       for (char&c: word) {
           if ('A' <= c && c <= 'Z') {
               c += 32;
           }
       }
       getline(cin,file);
       int lf = (int)file.size();
       
       string s = "";
       int cnt = 0;
       int pos = -1;
       char c;
       int start = 0;   // 当前单词的起始下标
       for (int i=0; i<=lf;i++) {
           // 遍历到边界，包含结尾虚拟空格
           c = (i<lf) ? file[i] : ' ';
           if ('A' <= c && c <= 'Z') {
               c += 32;
           }
           if (c==' ') {
               if (!s.empty() && s == word) {
                   cnt++;
                   // 记录初始位置
                   if (pos == -1) {
                       pos = start;
                   }
               }
               s.clear();
           } else {
               if (s.empty()) {
                   start = i; // 记录单词起始位置
               }
               s += c;
           }
       }
   
       if (cnt != 0) {
           cout << cnt << ' ' << pos;
       } else {
           cout << pos;
       }
       return 0;
   }
   ```

   

8. [         手机       ](https://www.luogu.com.cn/problem/P1765)(*)

   ```c++
   #include <iostream>
   #include <string>
   
   using namespace std;
   
   int main() {
       // 建立映射
       int key[128] = {0};
       string keys[] = {"abc","def","ghi","jkl",
           "mno","pqrs","tuv","wxyz"};
       // 记录字母对应按键次数
       for (int i=0;i<8;i++) {
           for (int j=0;j<keys[i].size();j++) {
               key[keys[i][j]] = 1+j;
           }
       }
       // 输入
       string str;
       getline(cin,str);
       int cnt = 0;
       for (char c: str) {
           if (c == ' ') {
               cnt++;
           } else {
               cnt += key[c];
           }
       }
       cout << cnt;
       return 0;
   }
   ```

   

9. [         小果的键盘       ](https://www.luogu.com.cn/problem/P3741)(*)

   ```c++
   #include <iostream>
   
   using namespace std;
   
   int countSon(string s, int len);
   
   int main() {
       int n;
       string s;
       cin >> n >> s;
       int maxn = countSon(s,n);
       for (char&c: s) {
           // 枚举只改变一个c
           char origin = c;
           c = origin=='V' ? 'K' : 'V';
           int cur = countSon(s,n);
           maxn = cur>maxn ? cur : maxn;
           c = origin;
       }
       cout << maxn << '\n';
       return 0;
   }
   
   // 统计子串数
   int countSon(string s, int len) {
       if (len <= 1) {
           return 0;
       }
       int cnt = 0;
       for (int i=0; i<len-1; i++) {
           if (s[i]=='V' && s[i+1]=='K') {
               cnt++;
           }
       }
       return cnt;
   }
   ```



10. [P1321     单词覆盖还原   ](https://www.luogu.com.cn/problem/P1321)(*)

    > 我有一个长度为 $l$ 的字符串，最开始时，这个字符串由 $l$ 个句号（`.`）组成。
    >
    > 我在这个字符串中，将多次把 `boy` 或者 `girl` 两单词，依次贴到这个字符串中。
    >
    > 后贴上单词，会覆盖之前贴上的单词，或者覆盖句号。最终，每个单词至少有一个字符没有被覆盖。
    >
    > 请问，一共贴有几个 `boy` 几个 `girl`？

    ```c++
    #include <iostream>
    #include <string>
    
     using namespace std;
    
    void concount(string s, int len,int&boy,int&girl);
    
    int main() { 
        string s;
        cin >> s;
        int len = s.size();
        int boy, girl;
        concount(s, len, boy, girl);
        cout << boy << '\n' << girl << '\n';
        return 0;
    }
    
    void concount(string s, int len,int&boy,int&girl) {
        boy=0,girl=0;
        for (int i=0; i<len; i++) {
            // 统计boy
            if (s[i] == 'b') {
                boy++;
            } else if (s[i]=='o' && (i==0 || s[i-1]!='b')) {
                boy++;
            } else if (s[i]=='y' && (i==0 || s[i-1]!='o')) {
                boy++;
            }
            // 统计girl
            if (s[i] == 'g') {
                girl++;
            } else if (s[i]=='i' && (i==0 || s[i-1]!='g')) {
                girl++;
            } else if (s[i]=='r' && (i==0 || s[i-1]!='i')) {
                girl++;
            } else if (s[i]=='l' && (i==0 || s[i-1]!='r')) {
                girl++;
            }
        }
    }
    ```

    


### 子串

1. [3. 无重复字符的最长子串](https://leetcode.cn/problems/longest-substring-without-repeating-characters/)(**)

   ```c++
   class Solution {
   public:
       int lengthOfLongestSubstring(string s) {
           int len = s.size();
           int ans = 0, l = 1;
           for (int i = 0; i < len; i++) {
               unordered_set<char> zcchar;
               l = 0;
               for (int j = i; j < len; j++) {
                   if (zcchar.find(s[j]) != zcchar.end()) {
                       ans = l > ans ? l : ans;
                       break;
                   } else {
                       zcchar.insert(s[j]);
                       l++;
                       ans = l > ans ? l : ans; // 更新最大长度
                   }
               }
           }
           return ans;
       }
   };
   ```

   > 双指针，滑窗

   ```c++
   class Solution {
   public:
       int lengthOfLongestSubstring(string s) {
           int len = s.size();
           int ans = 0, l = 0;
           unordered_set<char> zcchar;
           for (int i = 0, j = 0; i < len; i++) {
               // 滑窗，移动右指针
               for (; j < len; j++) {
                   if (zcchar.find(s[j]) != zcchar.end()) { // 找到重复
                       break;
                   } else {
                       zcchar.insert(s[j]);
                       l++;
                   }
               }
               ans = l > ans ? l : ans; // 更新最大长度
               l--;
               zcchar.erase(s[i]);
           }
           return ans;
       }
   };
   ```
