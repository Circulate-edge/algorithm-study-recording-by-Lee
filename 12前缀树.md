1. 类的前缀树实现

   ```c++
   // 前缀树
   class PriefixTree{
   public:
       class TreeNode {
       public:
           int pass; // 经过次数
           int end; // 结尾次数
           map<char, TreeNode*> next;
   
           TreeNode () {
               pass = 0;
               end = 0;
           }
   
           ~TreeNode() {
               for (auto& ct: next) {
                   delete ct.second;
               }
           }
       };
   
       PriefixTree() {
           root = new TreeNode(); // 树的根节点
       }
   
       void insert(string word) {
           TreeNode* node = root;
           node->pass++;
           for (int i = 0; i < (int)word.size(); i++) {
               if (!node->next->count(word[i])) { // 没有此分支
                   // 新建路径
                   node->next->insert({word[i],new TreeNode()});
               }
               node = node->next->at(word[i]);
               node->pass++;
           }
           node->end++;
       }
   
       void erase(string word) {
           if (countWords(word) > 0) { // 数词数
               TreeNode* node = root;
               TreeNode* next;
               node->pass--;
               for (int i = 0; i < word.size(); i++) {
                   next = node->next->at(word[i]); // 下一个树节点
                   if (--next->pass == 0) { 
                       // 删除节点及其分支
                       delete(node->next->at(word[i]));
                       node->next->erase(word[i]);
                       return;
                   }
                   node = next;
               }
               node->end--;
           }
       }
   
       int countWords(string word) {
           TreeNode *node = root;
           for (int i = 0; i < word.size(); i++) {
               if (node->next->find(word[i]) == node->next->end()) {
                   return 0;
               }
               node = node->next->at(word[i]);
           }
           return node->end;
       }
   
       int countPriefix(string word) {
           TreeNode *node = root;
           for (int i = 0; i < word.size(); i++) {
               if (node->next->find(word[i]) == node->next->end()) {
                   return 0;
               }
               node = node->next->at(word[i]);
           }
           return node->pass;
       }
   
       ~PriefixTree() {
           delete root;
       }
   private:
       TreeNode* root;
   };
   ```

1. [字典树的实现]([字典树的实现_牛客题霸_牛客网](https://www.nowcoder.com/practice/7f8a8553ddbf4eaab749ec988726702b))(**)

   ```c++
   #include <iostream>
   #include <algorithm>
   
   using namespace std;
   
   // 静态数组实现前缀树
   enum {
       MAXN = 150001
   };
   
   static int tree[MAXN][26] = {0};
   static int pass[MAXN] = {0};
   static int endCt[MAXN] = {0};
   static int cnt;
   
   int main() {
       void build();
       void insert(string word);
       int countWords(string word);
       int countPrefix(string prefix);
       void delWord(string word);
       void clear();
       build();
       int t, op;
       cin >> t;
       for (int i = 0; i < t; i++) {
           cin >> op;
           if (op == 1) {
               string word;
               cin >> word;
               insert(word);
           } else if (op == 2) {
               string word;
               cin >> word;
               delWord(word);
           } else if (op == 3) {
               string word;
               cin >> word;
               if (countWords(word) > 0) {
                   cout << "YES\n";
               } else {
                   cout << "NO\n";
               }
           } else if (op == 4) {
               string prefix;
               cin >> prefix;
               cout << countPrefix(prefix) << "\n";
           }
       }
       clear();
   }
   
   void build() {
       cnt = 1;
   }
   void insert(string word) {
       int cur = 1; // 根节点
       pass[cur]++;
       for (int i = 0, path; i < word.size(); i++) {
           path = word[i] - 'a'; // a~z对应0~25;
           if (tree[cur][path] == 0) { // 第0行不用，编号为0没节点
               tree[cur][path] = ++cnt;
           }
           cur = tree[cur][path]; // 跳转下一节点
           pass[cur]++; // 下一节点通过数+1
       }
       endCt[cur]++;
   }
   
   // 统计词数
   int countWords(string word) {
       int cur = 1;
       for (int i = 0, path; i < word.size(); i++) {
           path = word[i] - 'a';
           if (tree[cur][path] == 0) {
               return 0;
           }
           cur = tree[cur][path];
       }
       return endCt[cur];
   }
   
   // 统计前缀数
   int countPrefix(string prefix) {
       int cur = 1;
       for (int i = 0, path; i < prefix.size(); i++) {
           path = prefix[i] - 'a';
           if (tree[cur][path] == 0) {
               return 0;
           }
           cur = tree[cur][path];
       }
       return pass[cur];
   }
   
   void delWord(string word) {
       if (countWords(word) > 0) {
           int cur = 1;
           pass[cur]--;
           for (int i = 0, path; i < word.size(); i++) {
               path = word[i] - 'a'; 
               if (--pass[tree[cur][path]] == 0) { // 下一节点通过数为0
                   tree[cur][path] = 0; // 删除节点及其分支
                   return;
               }
               cur = tree[cur][path];
           }
           endCt[cur]--;
       }
   }
   
   void clear() {
       for (int i = 1; i <= cnt; i++) {
           fill(begin(tree[i]), end(tree[i]), 0);
           pass[i] = 0;
           endCt[i] = 0;
       }
   }
   ```

2. [接头密匙](https://www.nowcoder.com/practice/c552d3b4dfda49ccb883a6371d9a6932)(**)

   ```c++
   class Solution {
   public:
       /**
        * 牛牛和他的朋友们约定了一套接头密匙系统，用于确认彼此身份。
        * 密匙由一组数字序列表示，两个密匙被认为是一致的，
        * 如果满足以下条件：
        * 密匙 b 的长度不超过密匙 a 的长度。
        * 对于任意 0 <= i < length(b)，有 b[i+1] - b[i] == a[i+1] - a[i]。
        * 现在给定了m个密匙 b 的数组，以及n个密匙 a 的数组。
        * 请你返回一个长度为 m 的结果数组 ans，表示每个密匙b都有多少一致的密匙a。 
        * @param b int整型vector<vector<>> 
        * @param a int整型vector<vector<>> 
        * @return int整型vector
        */
       vector<int> countConsistentKeys(vector<vector<int> >& b, vector<vector<int> >& a) {
           // a的密钥序列化, 并插入前缀树
           string key;
           for (auto& nums: a) {
               key.clear();
               for (int i = 1; i < nums.size(); i++) {
                   key += to_string(nums[i] - nums[i - 1]) + '#';
               }
               // 插入前缀树
               insert(key);
           }
           // 在b中查找
           vector<int> ans(b.size());
           for (int i = 0; i < b.size(); i++) {
               key.clear();
               for (int j = 1; j < b[i].size(); j++) {
                   key += to_string(b[i][j] - b[i][j - 1]) + '#';
               }
               // 查找前缀数量
               ans[i] = countPrefix(key);
           }
           clear();
           return ans;
       }
   
       Solution() {
           cnt = 1;
       }
   
       void clear() {
           for (int i = 1; i <= cnt; i++) {
               for (int j = 0; j < 12; j++) {
                   tree[i][j] = 0;
               }
               pass[i] = 0;
               tail[i] = 0;
           }
           cnt = 1;
       }
   
       int pathof(const string& key, int i) {
           if (key[i] == '#')
               return 10; // 用10表示'#'
           else if (key[i] == '-') {
               return 11; // 用11表示'-'
           }
           return key[i] - '0';
       }
   
       void insert(const string& key) {
           int cur = 1; // 根节点
           pass[cur]++;
           for (int i = 0; i < key.size(); i++) {
               int path = pathof(key, i);
               // 没有的节点就新建
               if (tree[cur][path] == 0) {
                   tree[cur][path] = ++cnt;
               }
               cur = tree[cur][path];
               pass[cur]++;
           }
       }
   
       int countPrefix(const string& key) {
           int cur = 1;
           for (int i = 0; i < key.size(); i++) {
               int path = pathof(key, i);
               if (tree[cur][path] == 0) {
                   return 0;
               }
               cur = tree[cur][path];
           }
           return pass[cur];
       }
   
   private:
       enum {MAXN = 200001};
   
       inline static vector<vector<int>> tree = vector<vector<int>>(MAXN, vector<int>(12, 0));
       inline static vector<int> pass = vector<int>(MAXN, 0);
       inline static vector<int> tail = vector<int>(MAXN, 0);
   
       inline static int cnt;
   };
   ```

3. [421. 数组中两个数的最大异或值](https://leetcode.cn/problems/maximum-xor-of-two-numbers-in-an-array/)(**)

   > 前缀树
   >
   > 时间复杂度: $O(n * \log c)$ 
   >
   > 三重剪枝：
   >
   > 1. 前缀树指导递归展开，完成剪枝；
   > 2. 结束节点直接记录字符串，收集答案，不用再还原路径；
   > 3. 将收集过的节点pass值减1，收集过的分支不用再走，剪枝。

   ```c++
   class Solution {
   public:
       int findMaximumXOR(vector<int>& nums) {
           build(nums);
           int ans = 0;
           // 遍历nums，找出每个num分别与其他数两两异或, 取最大值
           for (int& num: nums) {
               ans = max(ans, maxXor(num));
           }
           clear();
           return ans;
       }
   
       enum {
           MAXN = 3000001
       };
   
       void insert(int num) {
           int cur = 1;
           for (int i = highbit, path; i >= 0; i--) {
               path = (num >> i) & 1;
               if (tree[cur][path] == 0) {
                   tree[cur][path] = ++cnt;
               }
               cur = tree[cur][path];
           }
       }
   
       void build(vector<int>& nums) {
           cnt = 1;
           int maxnum = 0;
           for (int i = 0; i < nums.size(); i++) {
               maxnum = max(maxnum, nums[i]);
           }
           highbit = getHigh(maxnum);
           for (int& num: nums) {
               // 插入前缀树
               insert(num);
           }
       }
   
       // num前缀树中数异或，可能的最大值
       int maxXor(int num) {
           int cur = 1, ans = 0;
           for (int i = highbit, status, want; i >= 0; i--) {
               status = (num >> i) & 1; // i位的状态
               want = status ^ 1; // i位的期望状态
               if (tree[cur][want] == 0) {
                   want ^= 1; // i位实际最大状态
               }
               ans |= (want ^ status) << i;
               cur = tree[cur][want];
           }
           return ans;
       }
   
       void clear() {
           for (int i = 1; i <= cnt; i++) {
               tree[i][0] = tree[i][1] = 0;
           }
       }
   
   private:
       // 第一个索引表示节点编号，第二个索引表示0或1的路径，值对应节点编号
       inline static vector<vector<int>> tree = vector<vector<int>>(MAXN, vector<int>(2, 0));
   
       int cnt;
   
       int highbit;
   
       int getHigh(int x) {
           int ans = 0;
           for (int i = 31; i >= 0; i--) {
               if (x & (1 << i)) {
                   ans = i;
                   break;
               }
           }
           return ans;
       }
   };
   ```

   > 哈希表
   >
   > 时间复杂度: $O(n * \log c)$ 

   ```c++
   #include <iostream>
   #include <vector>
   #include <unordered_set>
   
   using namespace std;
   
   class Solution {
   public:
       // 哈希表实现计算最大异或值
       int findMaximumXOR(vector<int>& nums) {
           int maxn = 0;
           for (int& num: nums) {
               maxn = max(maxn, num);
           }
           int highbit = gethighbit(maxn);
           int ans = 0;
           unordered_set<int> numset;
           for (int i = highbit; i >= 0; i--) {
               numset.clear();
               int better = ans | (1 << i);
               for (int& num: nums) {
                   // 取高位状态
                   int status = (num >> i) << i;
                   numset.insert(status);
                   // 在哈希表中查找是否存在能异或出更大值的数
                   if (numset.count(status ^ better)) {
                       ans = better;
                       break;
                   }
               }
           }
           return ans;
       }
   
       int gethighbit(int num) {
           for (int i = 31; i >= 0; i--) {
               if ((num >> i) & 1) {
                   return i;
               }
           }
           return 0;
       }
   };
   ```

4. [212. 单词搜索 II](https://leetcode.cn/problems/word-search-ii/)(***)

   > 前缀树，dfs

   ```c++
   #include <iostream>
   #include <vector>
   #include <array>
   
   using namespace std;
   
   class Solution {
   public:
       /**
        * 二维字符数组中，搜索位置连续的字符组成的单词。
        * 前缀树，dfs搜索，递归回溯
        */
       vector<string> findWords(vector<vector<char>>& board, vector<string>& words) {
           vector<string> ans;
           build(words);
           for (int i = 0; i < board.size(); i++) {
               for (int j = 0; j < board[0].size(); j++) {
                   // 遍历board，从每个位置的字符开始匹配dfs
                   dfs(board, ans, i, j, 1);
               }
           }
           clear();
           return ans;
       }
   
       enum {
           MAXN = 100001
       };
   
       // 把要查找的单词插入前缀树
       void build(vector<string>& words) {
           cnt = 1;
           for (string& str: words) {
               // 把字符串插入前缀树
               int cur = 1;
               pass[cur]++;
               for (int i = 0, path; i < str.size(); i++) {
                   path = str[i] - 'a';
                   if (tree[cur][path] == 0) {
                       tree[cur][path] = ++cnt;
                   }
                   cur = tree[cur][path];
                   pass[cur]++;
               }
               ends[cur] = str;
           }
       }
   
       // 从board的某个位置开始dfs,返回找到的单词数
       int dfs(vector<vector<char>>& board, vector<string>& ans, int i, int j, int cur) {
           // 越界或走重复的路，返回0
           if (i < 0 || i >= board.size() || j < 0 || j >= board[0].size() || board[i][j] == 0) {
               return 0;
           }
           char tmp =board[i][j]; // 临时记录当前字符
           int path = tmp - 'a';
           cur = tree[cur][path];
           if (pass[cur] == 0) { // cur == 0 || pass[cur] == 0当前分支查完了
               return 0;
           }
   
           // 若i,j位置的字符在前缀树中存在，继续递归
           int num = 0;
           if (ends[cur] != "") { // 找到一个单词
               num++;
               ans.push_back(ends[cur]);
               ends[cur] = ""; // 找到的删去避免重复添加
           }
           board[i][j] = 0; // 走过的路染色为0
           num += dfs(board, ans, i - 1, j, cur); // dfs上方
           num += dfs(board, ans, i + 1, j, cur); // dfs下方
           num += dfs(board, ans, i, j - 1, cur); // dfs左侧
           num += dfs(board, ans, i, j + 1, cur); // dfs右侧
           board[i][j] = tmp; // 回溯路径
           pass[cur] -= num; // 减去找到的单词数
           return num;
       }
   
       void clear() {
           for (int i = 1; i <= cnt; i++) {
               for (int j = 0; j < 26; j++) {
                   tree[i][j] = 0;
               }
               pass[i] = 0;
               ends[i] = "";
           }
       }
   
   private:
       inline static vector<array<int, 26>> tree = vector<array<int, 26>>(MAXN, {0});
       inline static vector<int> pass = vector<int>(MAXN, 0);
       inline static vector<string> ends = vector<string>(MAXN, "");
       int cnt = 1;
   };
   ```
