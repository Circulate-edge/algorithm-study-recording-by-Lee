   class Solution {
   public:
       /**
        * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
        *
        * 
        * @param s string字符串

       vector<string> generatePermutation(string s) {

           unordered_set<string> subset;

           vector<string> ans;

               ans.push_back(sub);
           }
           return ans;

       void gensub(const string&s, int i, int ls,
       unordered_set<string>&subset, string path) {
           // base case 
           if (i == ls) {
               // 添加路径字符串到集合
               subset.insert(path);
           } else {
               path.push_back(s[i]); // 添加i字符
               gensub(s, i+1, ls, subset, path); // 递归有i字符的
               path.pop_back();
               // 无i字符递归
               gensub(s, i+1, ls, subset, path);
           }
       }
   };

2. [78. 子集](https://leetcode.cn/problems/subsets/)(**)

   > 幂集

   ```c++
   class Solution {
   public:
       vector<vector<int>> subsets(vector<int>& nums) {
           vector<vector<int>> ans;
           // 递归子集
           int len = nums.size();
           vector<int> path;
           recursion(nums, ans, path, 0, len);
   
           return ans;
       }
   
       void recursion(vector<int>&nums, vector<vector<int>>&ans, vector<int>&path,
       int i, int len) {
           // base case
           if (i == len) {
               ans.push_back(path);
           } else {
               // 递归，2中分支，i加入了和i没加入的
               path.push_back(nums[i]);
               recursion(nums, ans, path, i+1, len);
               path.pop_back(); // 擦除路径
               recursion(nums, ans, path, i+1, len);
           }
       }
   };
   ```

   

3. [90. 子集 II](https://leetcode.cn/problems/subsets-ii/)(**)

   > 组合
   >
   > 1. 这是一种对二叉树的剪枝，因为组合只关心取几个数，不关心取的数位置是否相同；
   > 2. 当前数取0个的情况要和其他分开，因为取0个不会有添加当前数到路径的这一步；

   ```c++
   class Solution {
   public:
       vector<vector<int>> subsetsWithDup(vector<int>& nums) {
           // 返回数组可能组合
           vector<vector<int>> ans;
           // 排序数组
           sort(nums.begin(), nums.end());
           vector<int> path(nums.size()); // 路径数组
           combine(nums, ans, path, 0, nums.size(), 0);
           return ans;
       }
   
       void combine(vector<int>&nums, vector<vector<int>>&ans, vector<int>&path,
       int i, int len, int size) {
           if (i == len) { // base case
               // nums遍历完了
               vector<int> cur(size);
               for (int j=0; j<size; j++) {
                   cur[j] = path[j];
               }
               ans.push_back(cur);
   
           } else {
               // 找下一种数的起始索引j
               int j = i + 1;
               while (j<len && nums[i]==nums[j]) {
                   j++;
               }
   
               combine(nums, ans, path, j, len, size); // 当前数取0个
               for (; i < j; i++) {
                   // 当前数取n,n-1,...,1个
                   path[size++] = nums[i];
                   // 递归对下一个数选取的组合
                   combine(nums, ans, path, j, len, size);
               }
           }
       }
   };
   ```

   

4. [46. 全排列](https://leetcode.cn/problems/permutations/)(**)

   > 用递归来实现全排列，核心在于：
   >
   > 1. 在i位置换为i 到 n - 1 的元素，然后递归
   > 2. 递归后元素恢复为递归前的位置

   ```c++
   class Solution {
   public:
       vector<vector<int>> permute(vector<int>& nums) {
           vector<vector<int>> ans;
           int len = nums.size();
           // 全排列
           rangeRecursion(nums, len, 0, ans);
           return ans;
       }
   
       void rangeRecursion(vector<int>&nums,int len, int i, vector<vector<int>>&ans) {
           if (i == len) { // base case, 遍历完字符串
               vector<int> cur;
               for (int x: nums) {
                   cur.push_back(x);
               }
               ans.push_back(cur);
           } else {
               // 排列
               for (int j = i; j < len; j++) {
                   // 当前位置i为i以后的元素的不同情况
                   swap(nums, i, j);
                   rangeRecursion(nums, len, i+1, ans);
                   swap(nums, i, j); // 恢复数组
               }
           }
       }
   
       void swap(vector<int>&nums, int a, int b) {
           int temp = nums[a];
           nums[a] = nums[b];
           nums[b] = temp;
       }
   };
   ```

   

5. [47. 全排列 II](https://leetcode.cn/problems/permutations-ii/)(**)

   > 给定一个可包含重复数字的序列 `nums` ，***按任意顺序*** 返回所有不重复的全排列。

   > 1. 数组中会存在重复元素，交换i和j位置的数前检查j的数有没有到达过i;

   ```c++
   class Solution {
   public:
       vector<vector<int>> permuteUnique(vector<int>& nums) {
           vector<vector<int>> ans;
           int len = nums.size();
           // 去重全排列
           rprange(nums, len, 0, ans);
           return ans;
       }
   
       void rprange(vector<int>&nums, int len, int i, vector<vector<int>>&ans) {
           if (i == len) {
               // base case:添加当前路径到ans
               vector<int> cur;
               for (int x: nums) {
                   cur.push_back(x);
               }
               ans.push_back(cur);
           } else {
               // 用集合去重
               unordered_set<int> inum;
               for (int j = i; j < len; j++) {
                   // i后面的数交换到i位置
                   if (inum.find(nums[j]) == inum.end()) {
                       // j的数没来过i位置
                       inum.insert(nums[j]);
                       // i换位j的数，递归结束后擦除路径
                       swap(nums, i, j);
                       rprange(nums, len, i + 1, ans);
                       swap(nums, i, j);
                   }
               }
           }
       }
   
       void swap(vector<int>&nums, int a, int b) {
           int temp = nums[a];
           nums[a] = nums[b];
           nums[b] = temp;
       }
   };
   ```

6. [52. N 皇后 II](https://leetcode.cn/problems/n-queens-ii/)(数组实现)(***)

   > **n 皇后问题** 研究的是如何将 `n` 个皇后放置在 `n × n` 的棋盘上，并且使皇后彼此之间不能相互攻击。
   >
   > 给你一个整数 `n` ，返回 **n 皇后问题** 不同的解决方案的数量。

   > n皇后问题的经典实现，用数组记录路径，
   >
   > 时间复杂度 $O(n!)$; 常数时间较长
   >
   > 1. 判断*当前皇后是否在之前皇后攻击范围内*
   >
   > $$
   > |row_i - row_{i-n}| == |column_i - column_{i-n}|
   > $$
   >
   > 2. 改变数组表示每次决策，累加记录结果

   ```c++
   class Solution {
   public:
       // 经典方法实现
       int totalNQueens(int n) {
           if (n < 1) {
               return 0;
           }
           // 返回n皇后问题解的数量
           vector<int> col(n);
           return decursion(col, 0, n);
       }
   
       // 递归n皇后问题过程
       // i：当前来到的行; n: 行数
       int decursion(vector<int>& col, int i, int n) {
           // base case
           if (i == n) return 1;
           int ans = 0;
           // check这行的每个位置
           for (int j = 0; j < n; j++) {
               if (check(col,i,j)) {
                   col[i] = j;
                   ans += decursion(col, i + 1, n);
               }
           }
           return ans;
       }
   
       // 当前皇后位置在i行j列，检查会不会冲突
       bool check(vector<int>& col, int i, int j) {
           // check per queen before row i
           for (int k = 0; k < i; k++) {
               if (j == col[k] || (abs(i - k) == abs(j - col[k]))) {
                   return false;
               }
           }
           return true;
       }
   };
   ```

7. [52. N 皇后 II](https://leetcode.cn/problems/n-queens-ii/)(位运算实现)(***)

   > 1. 用位集存储皇后的位置信息，`col`的二进制为1的位就是有皇后的列
   > 2. `l` 表示之前皇后左下方对角线对这一行的影响范围，`r`表示之前皇后右下方对角线对这一行的影响范围，将`col`, `l`, `r` 用并位运算，得到`ban` ，二进制为1的就是不能放皇后的位置，`limit` 和`~ban` 与运算，得到`candidate`，二进制为1的就是这行能放皇后的位置，依次进行递归
   > 3. 用`lsb算法` 取`candiate` 最低有效位，二进制1表示这一行放皇后的列
   > 4. 改变`col`, `l`, `r` 的二进制状态表示每次决策，累加记录结果

   ```c++
   class Solution {
   public:
       int totalNQueens(int n) {
           if (n < 1) return 0;
           // 能放皇后的列为1
           // n = 5
   		// 1 << 5 = 0...100000 - 1
   		// limit  = 0...011111; 
           int limit = (1 << n) - 1;
           return dfs(limit, 0, 0, 0);
       }
   
       // col: 列的影响; l: 对左对角线的影响; r: 对右对角线的影响
       int dfs(int limit, int col, int l, int r) {
           // base case: 所有列都有皇后
           if (col == limit) return 1;
   
           int ans = 0;
           // 不能放皇后的位置(二进制1表示)
           int ban = col | l | r;
           // 能放皇后的位置(二进制1表示)
           int candidate = limit & ~(unsigned)ban;
           // 递归所有能放皇后的位置
           while (candidate > 0) {
               int place = candidate & (-candidate); // lowest significant bit
               ans += dfs(limit, col | place, (l | place) >> 1, (r | place) << 1);
               candidate ^= place; // 消去放过皇后的位置
           }
           return ans;
       }
   };
   ```

8. [51. N 皇后](https://leetcode.cn/problems/n-queens/)(***)

   > 按照国际象棋的规则，皇后可以攻击与之处在同一行或同一列或同一斜线上的棋子。
   >
   > **n 皇后问题** 研究的是如何将 `n` 个皇后放置在 `n×n` 的棋盘上，并且使皇后彼此之间不能相互攻击。
   >
   > 给你一个整数 `n` ，返回所有不同的 **n 皇后问题** 的解决方案。
   >
   > 每一种解法包含一个不同的 **n 皇后问题** 的棋子放置方案，该方案中 `'Q'` 和 `'.'` 分别代表了皇后和空位。

   > 1. 递归过程中，记录决策路径
   > 2. 每次递归后，擦除递归路径
   > 3. 要返回`Q` 的图案，预先把N皇后问题可能的行图案制成表，返回时直接用索引图案 
   > 4. place是二进制数，获取1的索引，才是皇后的索引位置

   ```c++
   class Solution {
   public:
       int solNum;
       vector<vector<string>> solveNQueens(int n) {
           vector<vector<string>> ans;
           vector<string> path;
           vector<string> rowmap(n);
           if (n < 1) {
               return ans;
           }
   
           for (int i = 0; i < n; i++) {
               rowmap[i] = genmap(i, n);
           }
           int limit = (1 << n) - 1;
           solNum = dfs(ans, path, rowmap, limit, 0, 0, 0);
           return ans;
       }
   
       int dfs(vector<vector<string>>&ans, vector<string>&path, vector<string>& rowmap, int limit, int col, int l, int r) {
           // 列已满
           if (col == limit) {
               ans.push_back(path);
               return 1;
           }
           int ban = col | l | r; // 本列皇后限制
           int candidate = ~ban & limit;
           int cnts = 0;
           // 所有可能位置
           while (candidate) {
               int place = candidate & -candidate; // lsb
               int bit = bitIndex(place);
               string cur = rowmap[bit];// 这一行的情况string
               path.push_back(cur);
               // 递归
               cnts += dfs(ans, path, rowmap, limit, col | place, (l | place) >> 1, (r | place) << 1);
               path.pop_back(); // 擦除路径
               candidate ^= place;
           }
           return cnts;
       }
   
       string genmap(int bit, int n) {
           return string(bit, '.') + "Q" + string(n - 1 - bit, '.');
       }
   
       // 输入二进制数，找到索引位
       int bitIndex(int place) {
           int bit = 0;
           while (place > 1) {
               place >>= 1;
               bit++;
           }
           return bit;
       }
   };
   ```

   

9. [消灭怪物](https://www.nowcoder.com/practice/d88ef50f8dab4850be8cd4b95514bbbd)(***)

   > 全排列
   >
   > 时间复杂度: $O(n!)$ 
   >
   > n小于等于11时，能在1秒内通过

   ```c++
   #include <iostream>
   #include <vector>
   #include <climits>
   
   using namespace std;
   
   enum {
       MAXN = 11
   };
   
   class Solution {
     public:
       /**
        * 最少技能数
        * @param n: 技能总数
        * @param m: 怪物血量
        */
       int lstOper(vector<int>& damage, vector<int>& blood, int n, int m) {
           // 返回答案
           int ans = t(damage, blood, n, m, 0);
           if (ans != INT_MAX) {
               return ans;
           } else {
               return -1;
           }
       }
   
       /**
        * 递归全排列,
        * @param damage: 伤害
        * @param blood: 暴击血量（双倍伤害）
        */
       int t(vector<int>& damage, vector<int>& blood, int n, int rest, int i) {
           // base case
           // monster had died
           if (rest <= 0) return i;
           // run out skills
           else if (i == n) return INT_MAX;
   
           // 至少几个技能可以杀死
           int ans = INT_MAX;
           // 在第i次攻击尝试 from i to n - 1 的技能
           for (int j = i; j < n; j++) {
               swap(damage, blood, i, j);
               // 本次伤害
               int cur = blood[i] < rest ? damage[i] : 2 * damage[i];
               // 递归i + 1个技能
               int tans = t(damage, blood, n, rest - cur, i + 1);
               if (tans < ans) ans = tans; // 更新答案
               swap(damage, blood, i, j); // 回溯
           }
           return ans;
       }
   
       void swap(vector<int>& damage, vector<int>& blood, int i, int j) {
           int tdmg, tbld;
           tdmg = damage[i], tbld = blood[i];
           damage[i] = damage[j], blood[i] = blood[j];
           damage[j] = tdmg, blood[j] = tbld;
       }
   };
   
   int main() {
       ios::sync_with_stdio(false);
       cin.tie(nullptr);
       Solution ls;
       int t, n, m;
       vector<int> damage(MAXN);
       vector<int> blood(MAXN);
       cin >> t;
       // t组数据
       for (int i = 0; i < t; i++) {
           cin >> n >> m;
           // n个技能
           for (int j = 0; j < n; j++) {
               cin >> damage[j] >> blood[j];
           }
           // 返回答案
           cout << ls.lstOper(damage, blood, n, m) << "\n";
       }
   }
   ```

1. [表达式求值](https://www.nowcoder.com/practice/c215ba61c8b1443b996351df929dc4d4)(***)

   > 基础计算器

   ```c++
   class Solution {
   public:
       /**
        * 代码中的类名、方法名、参数名已经指定，请勿修改，直接返回方法规定的值即可
        *
        * 返回表达式的值
        * @param s string字符串 待计算的表达式
        * @return int整型
        */
       int solve(string s) {
           // 初始化where
           where = 0;
           return calculate(s,0);
       }
   
       int where;
       // s[i....]开始计算，遇到字符串终止 或者 遇到)停止
       // 返回 : 自己负责的这一段，计算的结果
       // 返回之间，更新全局变量where，为了上游函数知道从哪继续！
       int calculate(string s, int i) {
           vector<int> nums;
           vector<char> signs;
           int cur = 0;
   
           while (i < s.size() && s[i] != ')') {
               if ('0' <= s[i] && s[i] <= '9') {
                   cur = cur * 10 + (s[i++] - '0');
               } else if (s[i] != '(') {
                   // 遇到运算符
                   push(nums, signs, cur, s[i++]);
                   cur = 0; // 压入数字后栈更新为0
               } else {
                   // 遇到(
                   cur = calculate(s, i + 1);
                   // 更新位置
                   i = where + 1;
               }
           }
           push(nums, signs, cur, '+');
           // 更新全局变量，让上层函数知道该从什么位置继续
           where = i;
           // 计算栈内的数
           return compute(nums, signs);
       }
   
       // 把数字和符号加入栈
       void push(vector<int>& nums, vector<char>& signs, int cur, char ops) {
           int n = nums.size();
   
           if (n == 0 || signs[n - 1] == '+' || signs[n - 1] == '-') {
               nums.push_back(cur);
               signs.push_back(ops);
           } else {
               // 遇到*号和/号结合
               int topnum = nums[n - 1];
               char topops = signs[n - 1];
               if (topops == '*') {
                   nums[n - 1] = topnum * cur;
               } else {
                   nums[n - 1] = topnum / cur;
               }
               signs[n - 1] = ops;
           }
       }
   
       // 计算只有+-的表达式
       int compute(vector<int>& nums, vector<char>& signs) {
           int n = nums.size();
           int ans = nums[0];
           for (int i = 1; i < n; i++) {
               if (signs[i - 1] == '+') {
                   ans += nums[i];
               } else {
                   ans -= nums[i];
               }
           }
           return ans;
       }
   };
   ```




2. [394. 字符串解码](https://leetcode.cn/problems/decode-string/)(**)

   ```c++
   class Solution {
   public:
       string decodeString(string s) {
           where = 0;
           return f(s, 0);
       }
   
       int where = 0;
   
       // 递归f(i): s[i~]
       string f(string s, int i) {
           string path = "";
           int cnt = 0;
           
           while (i < s.size() && s[i] != ']') {
               // 遇到结束条件停止
               if ('a' <= s[i] && s[i] <= 'z') {
                   path += s[i++];
               } else if (s[i] != '[') {
                   // 遇到数字，cnt开始累加次数
                   cnt = cnt * 10 + s[i++] - '0';
               } else {
                   // 遇到左括号
                   path += dec(f(s, i + 1), cnt);
                   i = where + 1;
                   cnt = 0; // 计算过cnt后重置
               }
           }
           where = i;
           return path;
       }
   
       // 对无嵌套的字符串解码
       string dec(string path, int cnt) {
           string ans = "";
           for (int i = 0; i < cnt; i++) {
               ans += path;
           }
           return ans;
       }
   };
   ```

3. [726. 原子的数量](https://leetcode.cn/problems/number-of-atoms/)(***)

   > 递归得到化学式原子的数量
   >
   > 1. 用有序表统计原子数量。循环过程中，遇到括号说明进入嵌套递归，遇到大写字母说明开始新元素，这2种都要把之前的历史记录填入有序表，然后使删除 `pre` 内存并使 `pre` 置空。
   > 2. 遇到小写字母则说明还不是新元素，接着添加到上一个字母后，
   > 3. 遇到数字则是进入统计。
   > 4. 递归后，函数结束后更新 `where` 变量，记录结束位置，在新递归完成后，依靠 `where` ，用 $i = where + 1$抵达下一轮轮循环 。

   ```c++
   class Solution {
   public:
       int where;
   
       string countOfAtoms (string formula) {
           where = 0;
           // 建有序表
           map<string, int>* atmap;
           // 求原子数
           atmap = new map<string, int>(cntatm(formula, 0));
           string ans = "";
   
           // 遍历有序表
           for (const auto& pair: *atmap) {
               ans += pair.first;
               int cnt = pair.second;
               if (cnt > 1) {
                   ans += to_string(cnt);
               }
           }
           delete atmap;
           return ans;
       }
   
       map<string, int> cntatm (string& s, int i) {
           map<string, int> atmap;
           // 元素名
           string name = "";
           // 历史有序表
           map<string, int>* pre = nullptr;
           int cnt = 0;
   
           while (i < (int)(s.size()) && s[i] != ')') {
               // 开始新元素，或新嵌套
               if (('A' <= s[i] && s[i] <= 'Z') || s[i] == '(') {
                   // 填有序表
                   fill(atmap, name, pre, cnt);
                   delete pre;
                   // 置空
                   name = "", pre = nullptr, cnt = 0;
                   if ('A' <= s[i] && s[i] <= 'Z') {
                       name += s[i++];
                   } else {
                       pre = new map<string, int>(cntatm(s, i + 1));
                       i = where + 1;
                   }
               // 小写字母，上一个元素
               } else if ('a' <= s[i] && s[i] <= 'z') {
                   name += s[i++];
               // 原子数
               } else if ('0' <= s[i] && s[i] <= '9') {
                   cnt = cnt * 10 + s[i++] - '0';
               } else {
                   // 不应发生的分支
                   i++;
               }
           }
           fill(atmap, name, pre, cnt);
           delete pre;
           where = i;
           return atmap;
       }
   
       void fill (map<string, int>& atmap, string& name, map<string, int>* pre, int cnt) {
           if (name.size() > 0 || pre != nullptr) {
               cnt = cnt == 0 ? 1 : cnt;
               string key;
               int val;
               // 历史元素名
               // 插入元素
               if (name.size() > 0) {
                   key = name;
                   // 原来有序表中有的元素加上
                   if (atmap.find(key) != atmap.end()) {
                       atmap[key] += cnt;
                   } else {
                       atmap.insert({key, cnt});
                   }
                   
               // 把历史有序表的结果插总有序表
               } else {
                   for (const auto& pair: *pre) {
                       key = pair.first;
                       val = pair.second;
                       if (atmap.find(key) != atmap.end()) {
                           atmap[key] += val * cnt;
                       } else {
                           atmap.insert({key, val * cnt});
                       }
                   }
               }
           }
   
       }
   };
   ```
